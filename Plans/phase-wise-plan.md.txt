// frontend/components/search/unified-search-bar.tsx
interface UnifiedSearchBarProps {
  placeholder?: string;
  searchFields?: string[];
  onSearch: (query: UnifiedSearchQuery) => void;
  onSuggestionClick: (suggestion: SearchSuggestion) => void;
  autoComplete?: boolean;
  recentSearches?: string[];
  popularSearches?: string[];
  showVoiceSearch?: boolean;
  className?: string;
}

export function UnifiedSearchBar({
  placeholder = "Search...",
  searchFields = ["name", "barcode", "sku"],
  onSearch,
  onSuggestionClick,
  autoComplete = true,
  recentSearches = [],
  popularSearches = [],
  showVoiceSearch = false,
  className,
}: UnifiedSearchBarProps) {
  const [query, setQuery] = useState('');
  const [suggestions, setSuggestions] = useState<SearchSuggestion[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [showVoice, setShowVoice] = useState(false);
  
  const debouncedSearch = useCallback(
    debounce((searchQuery: string) => {
      if (searchQuery.length >= 2) {
        loadSuggestions(searchQuery);
      } else {
        setSuggestions([]);
      }
    }, 300),
    []
  );
  
  const handleInputChange = (value: string) => {
    setQuery(value);
    debouncedSearch(value);
  };
  
  const handleSearch = () => {
    if (query.trim()) {
      const searchQuery: UnifiedSearchQuery = {
        q: query.trim(),
        searchFields,
        page: 1,
        pageSize: 20,
      };
      onSearch(searchQuery);
      setSuggestions([]);
    }
  };
  
  return (
    <div className={`relative ${className}`}>
      <div className="relative">
        <input
          type="text"
          value={query}
          onChange={(e) => handleInputChange(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && handleSearch()}
          placeholder={placeholder}
          className="w-full px-4 py-2 border rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
        />
        {showVoiceSearch && (
          <button
            onClick={() => setShowVoice(!showVoice)}
            className="absolute right-2 top-1/2 transform -translate-y-1/2"
          >
            <MicIcon className="w-5 h-5" />
          </button>
        )}
      </div>
      
      {/* Suggestions dropdown */}
      {suggestions.length > 0 && (
        <div className="absolute top-full mt-1 w-full bg-white border rounded-lg shadow-lg z-50">
          {suggestions.map((suggestion, index) => (
            <div
              key={index}
              onClick={() => onSuggestionClick(suggestion)}
              className="px-4 py-2 hover:bg-gray-50 cursor-pointer"
            >
              <div className="flex items-center">
                <SearchIcon className="w-4 h-4 mr-2" />
                <span>{suggestion.text}</span>
                {suggestion.type === 'recent' && (
                  <span className="text-xs text-gray-500 ml-auto">Recent</span>
                )}
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

interface FacetFiltersProps {
  facets: FacetResult[];
  selectedFacets: Record<string, string[]>;
  onFacetChange: (facet: string, value: string[], operation: 'add' | 'remove' | 'replace') => void;
  onFacetClear: (facet: string) => void;
  onClearAll: () => void;
  className?: string;
}

export function FacetFilters({
  facets,
  selectedFacets,
  onFacetChange,
  onFacetClear,
  onClearAll,
  className,
}: FacetFiltersProps) {
  const getFacetDisplayName = (facet: string) => {
    const facetConfig = facetDefinitions.find(f => f.name === facet);
    return facetConfig?.label || facet;
  };
  
  const getFacetValuesDisplay = (facet: FacetResult) => {
    return facet.values.map(value => ({
      value: value.value,
      count: value.count,
      percentage: facet.total
        ? Math.round((value.count / facet.total) * 100)
        : 0,
    }));
  };
  
  return (
    <div className={`bg-white border rounded-lg p-4 ${className}`}>
      <div className="flex justify-between items-center mb-4">
        <h3 className="text-lg font-semibold">Filters</h3>
        {hasActiveFilters(selectedFacets) && (
          <button
            onClick={onClearAll}
            className="text-sm text-blue-600 hover:text-blue-800"
          >
            Clear all
          </button>
        )}
      </div>
      
      {facets.map(facet => (
        <div key={facet.name} className="mb-4">

```typescript
// frontend/components/filters/advanced-filters.tsx
interface AdvancedFiltersProps {
  filterSchema?: FilterSchema;
  onFiltersChange: (filters: FilterCondition[]) => void;
  className?: string;
}

export function AdvancedFilters({
  filterSchema,
  onFiltersChange,
  className,
}: AdvancedFiltersProps) {
  const [filters, setFilters] = useState<FilterCondition[]>([]);
  const [availableFields, setAvailableFields] = useState<string[]>([]);
  
  useEffect(() => {
    if (filterSchema) {
      setAvailableFields(filterSchema.fields.map(f => f.name));
    }
  }, [filterSchema]);
  
  const addFilter = () => {
    const newFilter: FilterCondition = {
      field: '',
      operator: 'eq',
      value: null,
    };
    setFilters([...filters, newFilter]);
  };
  
  const updateFilter = (index: number, updates: Partial<FilterCondition>) => {
    const newFilters = [...filters];
    newFilters[index] = { ...newFilters[index], ...updates };
    setFilters(newFilters);
    onFiltersChange(newFilters);
  };
  
  const removeFilter = (index: number) => {
    const newFilters = filters.filter((_, i) => i !== index);
    setFilters(newFilters);
    onFiltersChange(newFilters);
  };
  
  return (
    <div className={`bg-white border rounded-lg p-4 ${className}`}>
      <div className="flex justify-between items-center mb-4">
        <h3 className="text-lg font-semibold">Advanced Filters</h3>
        <button
          onClick={addFilter}
          className="px-3 py-1 bg-blue-600 text-white rounded hover:bg-blue-700"
        >
          Add Filter
        </button>
      </div>
      
      {filters.map((filter, index) => (
        <div key={index} className="flex gap-2 mb-2">
          <select
            value={filter.field}
            onChange={(e) => updateFilter(index, { field: e.target.value })}
            className="px-2 py-1 border rounded"
          >
            <option value="">Select field</option>
            {availableFields.map(field => (
              <option key={field} value={field}>{field}</option>
            ))}
          </select>
          
          <select
            value={filter.operator}
            onChange={(e) => updateFilter(index, { operator: e.target.value as any })}
            className="px-2 py-1 border rounded"
          >
            <option value="eq">Equals</option>
            <option value="ne">Not Equals</option>
            <option value="gt">Greater Than</option>
            <option value="gte">Greater Than or Equal</option>
            <option value="lt">Less Than</option>
            <option value="lte">Less Than or Equal</option>
            <option value="like">Contains</option>
            <option value="in">In List</option>
            <option value="between">Between</option>
          </select>
          
          <input
            type="text"
            value={filter.value || ''}
            onChange={(e) => updateFilter(index, { value: e.target.value })}
            placeholder="Value"
            className="px-2 py-1 border rounded"
          />
          
          <button
            onClick={() => removeFilter(index)}
            className="px-2 py-1 bg-red-600 text-white rounded hover:bg-red-700"
          >
            Remove
          </button>
        </div>
      ))}
    </div>
  );
}

// frontend/components/sorting/sort-dropdown.tsx
interface SortDropdownProps {
  sortOptions: SortOption[];
  currentSort: { field: string; direction: 'asc' | 'desc' };
  onSortChange: (sort: { field: string; direction: 'asc' | 'desc' }) => void;
  className?: string;
}

export function SortDropdown({
  sortOptions,
  currentSort,
  onSortChange,
  className,
}: SortDropdownProps) {
  const [isOpen, setIsOpen] = useState(false);
  
  const getSortLabel = (field: string) => {
    const option = sortOptions.find(opt => opt.field === field);
    return option?.label || field;
  };
  
  const toggleSortDirection = () => {
    onSortChange({
      ...currentSort,
      direction: currentSort.direction === 'asc' ? 'desc' : 'asc',
    });
  };
  
  return (
    <div className={`relative ${className}`}>
      <button
        onClick={() => setIsOpen(!isOpen)}
        className="flex items-center gap-2 px-3 py-2 border rounded hover:bg-gray-50"
      >
        <SortIcon className="w-4 h-4" />
        <span>{getSortLabel(currentSort.field)}</span>
        <span className="text-gray-500">
          ({currentSort.direction === 'asc' ? '↑' : '↓'})
        </span>
        <ChevronDownIcon className="w-4 h-4" />
      </button>
      
      {isOpen && (
        <div className="absolute top-full mt-1 w-64 bg-white border rounded shadow-lg z-50">
          {sortOptions.map((option) => (
            <button
              key={option.field}
              onClick={() => {
                onSortChange({
                  field: option.field,
                  direction: option.defaultDirection || 'desc',
                });
                setIsOpen(false);
              }}
              className="w-full text-left px-3 py-2 hover:bg-gray-50"
            >
              <div className="flex justify-between items-center">
                <span>{option.label}</span>
                {option.field === currentSort.field && (
                  <span className="text-blue-600">
                    {currentSort.direction === 'asc' ? '↑' : '↓'}
                  </span>
                )}
              </div>
            </button>
          ))}
        </div>
      )}
    </div>
  );
}
```
